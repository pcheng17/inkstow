  set(DESKTOP_WINDOW_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/DesktopWindowWindows.cpp")

  set(DESKTOP_WINDOW_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/DesktopWindowLinux.cpp")
